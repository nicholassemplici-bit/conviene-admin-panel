"""CONVIENE - verifica in sola lettura della copertura Open Prices.""" import json import os import time import urllib.error import urllib.parse import urllib.request from collections import Counter from datetime import date, timedelta SUPABASE_URL = os.environ['SUPABASE_URL'].rstrip('/') SUPABASE_KEY = os.environ['SUPABASE_SECRET_KEY'] LIMIT = min(max(int(os.getenv('CONVIENE_EAN_LIMIT', '200')), 1), 1000) BATCH_SIZE = 20 API = 'https://prices.openfoodfacts.org/api/v1/prices' HEADERS = {'User-Agent': 'CONVIENE/0.2 (GitHub catalog importer; read-only coverage check)', 'Accept': 'application/json'} def get_json(url, headers, attempts=3): for attempt in range(1, attempts + 1): try: req = urllib.request.Request(url, headers=headers) with urllib.request.urlopen(req, timeout=45) as response: return json.load(response) except urllib.error.HTTPError as exc: code = exc.code detail = exc.read(200).decode('utf-8', 'replace') if code not in (429, 500, 502, 503, 504) or attempt == attempts: raise RuntimeError(f'HTTP {code} da {urllib.parse.urlsplit(url).netloc}: {detail[:100]}') from exc print(f'HTTP {code}, tentativo {attempt}/{attempts}; pausa...', flush=True) except (urllib.error.URLError, TimeoutError) as exc: if attempt == attempts: raise RuntimeError(f'Connessione non riuscita: {exc}') from exc time.sleep(5 * attempt) raise RuntimeError('Richiesta non riuscita') def get_eans(): params = urllib.parse.urlencode({'select': 'ean', 'order': 'id.asc', 'limit': str(LIMIT)}) rows = get_json(f'{SUPABASE_URL}/rest/v1/product_variants?{params}', { 'apikey': SUPABASE_KEY, 'Accept': 'application/json', }) return list(dict.fromkeys(str(r.get('ean') or '').strip() for r in rows if r.get('ean'))) def main(): print('CONVIENE - OPEN PRICES - SOLO SIMULAZIONE', flush=True) print('Nessuna modifica a Supabase. Nessun prezzo viene registrato.', flush=True) eans = get_eans() print(f'EAN del catalogo selezionati: {len(eans)}', flush=True) if not eans: raise RuntimeError('Nessun EAN trovato') cutoff = (date.today() - timedelta(days=90)).isoformat() metrics = Counter() samples = [] for start in range(0, len(eans), BATCH_SIZE): group = eans[start:start + BATCH_SIZE] page = 1 while page <= 10: query = urllib.parse.urlencode({ 'product_code__in': ','.join(group), 'currency': 'EUR', 'type': 'PRODUCT', 'date__gte': cutoff, 'size': 100, 'page': page, }) data = get_json(API + '?' + query, HEADERS) items = data.get('items', []) if not isinstance(items, list): raise RuntimeError('Formato inatteso: items non e una lista') metrics['risultati_europei'] += len(items) for item in items: if item.get('product_code') not in group: continue loc = item.get('location') or {} country = str(loc.get('osm_address_country_code') or '').lower() if country != 'it': metrics['fuori_italia_o_senza_paese'] += 1 continue if item.get('duplicate_of') is not None: metrics['duplicati'] += 1 continue if item.get('price') is None or not item.get('date'): metrics['dati_incompleti'] += 1 continue metrics['italia_validi'] += 1 if len(samples) < 15: samples.append({ 'ean': item.get('product_code'), 'prezzo_eur': item.get('price'), 'data': item.get('date'), 'negozio': loc.get('osm_name') or loc.get('osm_brand'), 'citta': loc.get('osm_address_city'), 'id_prezzo': item.get('id'), }) pages = data.get('pages', 1) if not isinstance(pages, int) or page >= pages: break page += 1 metrics['ean_esaminati'] += len(group) print(f"Esaminati {metrics['ean_esaminati']}/{len(eans)} EAN - prezzi italiani recenti: {metrics['italia_validi']}", flush=True) time.sleep(1) print('\n===== RIEPILOGO OPEN PRICES =====') print(f"EAN esaminati: {metrics['ean_esaminati']}") print(f"Prezzi EUR ultimi 90 giorni restituiti: {metrics['risultati_europei']}") print(f"Prezzi con negozio in Italia: {metrics['italia_validi']}") print(f"Prezzi fuori Italia o senza paese: {metrics['fuori_italia_o_senza_paese']}") print('Esempi (non importati):') for s in samples: print(json.dumps(s, ensure_ascii=False)) print('SIMULAZIONE COMPLETATA: NESSUNA SCRITTURA SU SUPABASE') print('Attenzione: la presenza in Italia non garantisce presenza nelle province MI/PV/CO.') if __name__ == '__main__': main()
